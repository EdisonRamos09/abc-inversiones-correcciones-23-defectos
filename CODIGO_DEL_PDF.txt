function parseAmount(raw: string): string {
  const clean = raw.replace(/[^\d.,]/g, "");
  const thousands = /^\d{1,3}(,\d{3})+(\.\d{1,2})?$/.test(clean);
  const value = thousands ? clean.replace(/,/g, "") : clean.replace(",", ".");
  return Number(value).toFixed(2);
}
